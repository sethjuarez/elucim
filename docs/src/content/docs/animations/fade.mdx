Use opacity timelines for fade motion in normalized documents. Do not author `FadeIn`, `FadeOut`, or `fadeIn` props in `ElucimDocument`; document validation rejects wrapper animation syntax.

## Fade in

```ts
timelines: {
  intro: {
    id: 'intro',
    duration: 30,
    tracks: [
      {
        target: 'circle',
        property: 'opacity',
        keyframes: [
          { frame: 0, value: 0 },
          { frame: 30, value: 1, easing: 'easeOutCubic' },
        ],
      },
    ],
  },
}
```

## Fade out

```ts
timelines: {
  outro: {
    id: 'outro',
    duration: 30,
    tracks: [
      {
        target: 'circle',
        property: 'opacity',
        keyframes: [
          { frame: 0, value: 1 },
          { frame: 30, value: 0, easing: 'easeInCubic' },
        ],
      },
    ],
  },
}
```

## Complete scene

```tsx
import { DslRenderer, type ElucimDocument } from '@elucim/dsl';

const doc: ElucimDocument = {
  version: '2.0',
  scene: { type: 'player', width: 500, height: 250, children: ['circle'] },
  elements: {
    circle: {
      id: 'circle',
      type: 'circle',
      props: { type: 'circle', cx: 250, cy: 125, r: 50, stroke: '$accent', strokeWidth: 3, fill: 'none', opacity: 0 },
    },
  },
  timelines: {
    intro: {
      id: 'intro',
      duration: 30,
      tracks: [{ target: 'circle', property: 'opacity', keyframes: [{ frame: 0, value: 0 }, { frame: 30, value: 1 }] }],
    },
  },
  defaultStateMachine: 'main',
  stateMachines: {
    main: {
      id: 'main',
      entry: 'intro',
      states: { intro: { timeline: 'intro' } },
      transitions: [{ id: 'entry-start', from: 'entry', to: 'intro', trigger: 'onStart' }],
    },
  },
};

export function FadeScene() {
  return <DslRenderer dsl={doc} />;
}
```
